eyes_refresh_country_list = {
	if = {
		limit = { has_country_flag = eyes_gui_refresh_diagnostics_active }
		set_variable = { var = global.eyes_gui_country_count value = 0 }
		set_variable = { var = global.eyes_gui_array_stored_count value = 0 }
		set_variable = { var = global.eyes_gui_player_array_stored_count value = 0 }
		set_variable = { var = global.eyes_gui_dynamic_probe_count value = 0 }
		log = "EYES_HOI4_COUNTRY_REFRESH_BEGIN_V1|[ROOT.GetTag]"
	}
	clear_array = global.eyes_gui_dynamic_probe_values
	add_to_array = { array = global.eyes_gui_dynamic_probe_values value = 1 }
	if = {
		limit = {
			has_country_flag = eyes_gui_refresh_diagnostics_active
			is_in_array = { array = global.eyes_gui_dynamic_probe_values value = 1 }
		}
		add_to_variable = { var = global.eyes_gui_dynamic_probe_count value = 1 }
		log = "EYES_HOI4_DYNAMIC_PROBE_ARRAY_STORED_V1|[ROOT.GetTag]|1"
	}
	if = {
		limit = {
			has_country_flag = eyes_gui_refresh_diagnostics_active
			NOT = { is_in_array = { array = global.eyes_gui_dynamic_probe_values value = 1 } }
		}
		log = "EYES_HOI4_DYNAMIC_PROBE_ARRAY_MISSING_V1|[ROOT.GetTag]|1"
	}
	if = {
		limit = { has_country_flag = eyes_gui_refresh_diagnostics_active }
		log = "EYES_HOI4_DYNAMIC_PROBE_ARRAY_READY_V1|[ROOT.GetTag]|1|[?global.eyes_gui_dynamic_probe_count]"
	}
	clear_array = global.eyes_available_countries
	clear_array = ROOT.eyes_player_available_countries
	every_country = {
		limit = {
			exists = yes
			any_owned_state = { always = yes }
			NOT = { tag = ROOT }
		}
		add_to_array = { array = global.eyes_available_countries value = THIS }
		add_to_array = { array = ROOT.eyes_player_available_countries value = THIS }
		if = {
			limit = { ROOT = { has_country_flag = eyes_gui_refresh_diagnostics_active } }
			add_to_variable = { var = global.eyes_gui_country_count value = 1 }
			log = "EYES_HOI4_COUNTRY_CANDIDATE_V1|[ROOT.GetTag]|[THIS.GetTag]"
		}
		if = {
			limit = {
				ROOT = { has_country_flag = eyes_gui_refresh_diagnostics_active }
				is_in_array = { array = global.eyes_available_countries value = THIS }
			}
			add_to_variable = { var = global.eyes_gui_array_stored_count value = 1 }
			log = "EYES_HOI4_COUNTRY_ARRAY_STORED_V1|[ROOT.GetTag]|[THIS.GetTag]"
		}
		if = {
			limit = {
				ROOT = { has_country_flag = eyes_gui_refresh_diagnostics_active }
				is_in_array = { array = ROOT.eyes_player_available_countries value = THIS }
			}
			ROOT = { add_to_variable = { var = global.eyes_gui_player_array_stored_count value = 1 } }
			log = "EYES_HOI4_COUNTRY_PLAYER_ARRAY_STORED_V1|[ROOT.GetTag]|[THIS.GetTag]"
		}
		if = {
			limit = {
				ROOT = { has_country_flag = eyes_gui_refresh_diagnostics_active }
				NOT = { is_in_array = { array = ROOT.eyes_player_available_countries value = THIS } }
			}
			log = "EYES_HOI4_COUNTRY_PLAYER_ARRAY_MISSING_V1|[ROOT.GetTag]|[THIS.GetTag]"
		}
		if = {
			limit = {
				ROOT = { has_country_flag = eyes_gui_refresh_diagnostics_active }
				NOT = { is_in_array = { array = global.eyes_available_countries value = THIS } }
			}
			log = "EYES_HOI4_COUNTRY_ARRAY_MISSING_V1|[ROOT.GetTag]|[THIS.GetTag]"
		}
	}
	if = {
		limit = { has_country_flag = eyes_gui_refresh_diagnostics_active }
		log = "EYES_HOI4_COUNTRY_REFRESH_END_V1|[ROOT.GetTag]|[?global.eyes_gui_country_count]|[?global.eyes_gui_array_stored_count]"

	}
	eyes_rebuild_country_page = yes
}

eyes_rebuild_country_page = {
	clear_array = global.eyes_visible_countries
	set_variable = { var = ROOT.eyes_gui_page_count value = 1 }
	set_variable = { var = ROOT.eyes_gui_page_boundary value = 9 }

	# Count pages from zero-based array indices: 0-9 is page one, 10-19 page two, etc.
	for_each_loop = {
		array = global.eyes_available_countries
		index = eyes_gui_page_index
		value = eyes_gui_page_country
		if = {
			limit = {
				check_variable = { eyes_gui_page_index > ROOT.eyes_gui_page_boundary }
			}
			add_to_variable = { var = ROOT.eyes_gui_page_count value = 1 }
			add_to_variable = { var = ROOT.eyes_gui_page_boundary value = 10 }
		}
	}

	set_variable = { var = ROOT.eyes_gui_page_last value = ROOT.eyes_gui_page_count }
	subtract_from_variable = { var = ROOT.eyes_gui_page_last value = 1 }
	if = {
		limit = { check_variable = { ROOT.eyes_gui_page > ROOT.eyes_gui_page_last } }
		set_variable = { var = ROOT.eyes_gui_page value = ROOT.eyes_gui_page_last }
	}
	if = {
		limit = { check_variable = { ROOT.eyes_gui_page < 0 } }
		set_variable = { var = ROOT.eyes_gui_page value = 0 }
	}

	set_variable = { var = ROOT.eyes_gui_page_start value = ROOT.eyes_gui_page }
	multiply_variable = { var = ROOT.eyes_gui_page_start value = 10 }
	set_variable = { var = ROOT.eyes_gui_page_end value = ROOT.eyes_gui_page_start }
	add_to_variable = { var = ROOT.eyes_gui_page_end value = 10 }
	set_variable = { var = ROOT.eyes_gui_page_display value = ROOT.eyes_gui_page }
	add_to_variable = { var = ROOT.eyes_gui_page_display value = 1 }

	for_each_loop = {
		array = global.eyes_available_countries
		index = eyes_gui_page_index
		value = eyes_gui_page_country
		if = {
			limit = {
				check_variable = {
					var = eyes_gui_page_index
					value = ROOT.eyes_gui_page_start
					compare = greater_than_or_equals
				}
				check_variable = { eyes_gui_page_index < ROOT.eyes_gui_page_end }
			}
			add_to_array = {
				array = global.eyes_visible_countries
				value = var:eyes_gui_page_country
			}
		}
	}
}

eyes_ensure_campaign_identity = {
	if = {
		limit = { NOT = { has_global_flag = eyes_campaign_identity_seeded } }
		# These values persist in the save. Bridge derives the campaign UUID from
		# this token; no static campaign ID is shared by separate campaigns.
		set_variable_to_random = { var = global.eyes_campaign_seed_1 min = 1 max = 2147483647 integer = yes }
		set_variable_to_random = { var = global.eyes_campaign_seed_2 min = 1 max = 2147483647 integer = yes }
		set_variable_to_random = { var = global.eyes_campaign_seed_3 min = 1 max = 2147483647 integer = yes }
		set_variable_to_random = { var = global.eyes_campaign_seed_4 min = 1 max = 2147483647 integer = yes }
		set_global_flag = eyes_campaign_identity_seeded
	}
}
